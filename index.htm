        // Tab Switcher Logic

        function switchTab(tab) {

            const physicalTab = document.getElementById('tab-physical');

            const postalTab = document.getElementById('tab-postal');

            const physicalBtn = document.getElementById('tab-physical-btn');

            const postalBtn = document.getElementById('tab-postal-btn');

 

            if (tab === 'physical') {

                physicalTab.classList.remove('hidden');

                postalTab.classList.add('hidden');

               

                physicalBtn.className = "flex-1 py-2 px-3 rounded-lg text-center transition-all bg-blue-600 text-white font-semibold shadow";

                postalBtn.className = "flex-1 py-2 px-3 rounded-lg text-center text-slate-400 hover:text-slate-200 transition-all";

            } else {

                physicalTab.classList.add('hidden');

                postalTab.classList.remove('hidden');

 

                postalBtn.className = "flex-1 py-2 px-3 rounded-lg text-center transition-all bg-blue-600 text-white font-semibold shadow";

                physicalBtn.className = "flex-1 py-2 px-3 rounded-lg text-center text-slate-400 hover:text-slate-200 transition-all";

            }

        }

 

        // Generate vCard Data

        function downloadVCard() {

            const vCardData = `BEGIN:VCARD

VERSION:3.0

FN:Nickolin Naidoo

N:Naidoo;Nickolin;;;

ORG:AVM Consulting (Professional and Chartered Accountants S.A)

TITLE:Manager

NOTE:Specializations: Audit-Assurance, Tax, Advisory

TEL;TYPE=CELL:0718844662

TEL;TYPE=WORK,FAX:0333451224

ADR;TYPE=WORK:;;80 Chase Valley Road, Chase Valley;Pietermaritzburg;;3201;South Africa

POSTAL;TYPE=WORK:;;PO Box 400;Msunduzi;;3231;South Africa

END:VCARD`;

 

            const blob = new Blob([vCardData], { type: 'text/vcard;charset=utf-8;' });

            const url = URL.createObjectURL(blob);

            const link = document.createElement('a');

            link.href = url;

            link.setAttribute('download', 'Nickolin_Naidoo_AVM_Consulting.vcf');

            document.body.appendChild(link);

            link.click();

            document.body.removeChild(link);

           

            showToast('vCard contact file downloaded!', 'fa-file-arrow-down', 'text-blue-400');

        }

 

        // Web Share API with Clipboard Fallback

        async function shareCard() {

            const shareData = {

                title: 'Nickolin Naidoo - Digital Business Card',

                text: 'Contact details for Nickolin Naidoo, Manager at AVM Consulting.',

                url: window.location.href

            };

 

            if (navigator.share) {

                try {

                    await navigator.share(shareData);

                } catch (err) {

                    if (err.name !== 'AbortError') {

                        copyLinkToClipboard();

                    }

                }

            } else {

                copyLinkToClipboard();

            }

        }

 

        // Copy Link to Clipboard

        function copyLinkToClipboard() {

            const dummy = document.createElement('input');

            document.body.appendChild(dummy);

            dummy.value = window.location.href;

            dummy.select();

            document.execCommand('copy');

            document.body.removeChild(dummy);

            showToast('Digital Card link copied to clipboard!', 'fa-circle-check', 'text-emerald-400');

        }

 

        // Modal Toggle Logic

        function toggleModal(id, show) {

            const modal = document.getElementById(id);

            const card = document.getElementById(id + '-card');

           

            if (show) {

                modal.classList.remove('opacity-0', 'pointer-events-none');

                modal.classList.add('opacity-100', 'pointer-events-auto');

                if (card) {

                    card.classList.remove('scale-95');

                    card.classList.add('scale-100');

                }

            } else {

                modal.classList.remove('opacity-100', 'pointer-events-auto');

                modal.classList.add('opacity-0', 'pointer-events-none');

                if (card) {

                    card.classList.remove('scale-100');

                    card.classList.add('scale-95');

                }

            }

        }

 

        // Toast Popup Helper

        function showToast(message, iconClass = 'fa-circle-check', iconColor = 'text-emerald-400') {

            const toast = document.getElementById('toast');

            const toastMsg = document.getElementById('toast-msg');

            const toastIcon = document.getElementById('toast-icon');

 

            toastMsg.innerText = message;

            toastIcon.className = `fa-solid ${iconClass} ${iconColor} text-sm`;

 

            toast.classList.remove('opacity-0', 'pointer-events-none');

            toast.classList.add('opacity-100');

 

            setTimeout(() => {

                toast.classList.remove('opacity-100');

                toast.classList.add('opacity-0', 'pointer-events-none');

            }, 3000);

        }
